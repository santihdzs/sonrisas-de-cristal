\section{Demo en vivo (5 min)}

\begin{frame}{Demo en vivo \hfill\small 5 min}
  \large
  \begin{enumerate}
    \item APK \textit{release} instalada en un teléfono
    \item Educadora: registra la llegada, la bitácora y una foto
    \item Familia (otra cuenta, mismo backend): ve el día de su hijo
    \item Familia y educadora: intercambian un mensaje
    \item Fundación: desbloquea una cuenta bloqueada
  \end{enumerate}
