\section{Capacidad diferenciadora (1 min)}

\begin{frame}{Fotos con reglas \hfill\small 1 min}
  \large
  \begin{itemize}
    \item Dos permisos de imagen separados: actividades internas y redes sociales
    \item Cada familia ve solo las fotos de su hijo
    \item Cualquiera de esas familias puede borrar la foto para todos
    \item Cada captura y cada consulta quedan registradas
  \end{itemize}
\end{frame}
