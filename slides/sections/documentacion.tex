\section{Documentación, implantación y dilemas éticos (3 min)}

\begin{frame}{Documentación e implantación \hfill\small 3 min}
  \large
  \begin{itemize}
    \item Manual de usuario, manual de instalación y arquitectura
    \item APK firmada y backend en Docker
    \item Sin datos previos que migrar: la bitácora en papel se reemplaza por captura inicial desde la app
    \item \textit{Rollback} con \texttt{pg\_dump}
  \end{itemize}
\end{frame}

\begin{frame}{Aviso de privacidad}
  \large
  \begin{itemize}
    \item LFPDPPP (DOF 20-03-2025): arts. 8, 15, 21--31 y 36
    \item LGDNNA: art. 76
  \end{itemize}
\end{frame}
